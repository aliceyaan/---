--2. Создание таблиц и связей

--ЧАСТЬ 2. ЗАПРОСЫ
--1.1. неявное соединение таблиц с фильтром данных (WHERE) и сортировкой (ORDER BY)
SELECT
    r.kod_zayavki,
    r.tema_zayavki,
    e.fio AS initsiator,
    rt.nazvanie_tipa_zayavki AS tip,
    p.nazvanie_prioriteta AS prioritet,
    s.nazvanie_statusa AS status,
    r.data_sozdaniya,
    r.planovyy_srok
FROM requests r, employees e, request_types rt, priorities p, statuses s
WHERE r.kod_initsiatora = e.kod_sotrudnika
  AND r.kod_tipa_zayavki = rt.kod_tipa_zayavki
  AND r.kod_prioriteta = p.kod_prioriteta
  AND r.kod_statusa = s.kod_statusa
  AND p.kod_prioriteta IN (3, 4)
ORDER BY r.data_sozdaniya DESC;

SELECT
    e.kod_sotrudnika,
    e.fio,
    pos.nazvanie_dolzhnosti,
    d.nazvanie_podrazdeleniya,
    e.email,
    e.telefon
FROM employees e, positions pos, departments d
WHERE e.kod_dolzhnosti = pos.kod_dolzhnosti
  AND e.kod_podrazdeleniya = d.kod_podrazdeleniya
  AND e.status_sotrudnika = 'активен'
ORDER BY d.nazvanie_podrazdeleniya, e.fio;

--1.2. явное соединение таблиц (JOIN)
SELECT
    t.kod_zadachi,
    t.naimenovanie_zadachi,
    isp.fio AS ispolnitel,
    post.fio AS postanovshchik,
    s.nazvanie_statusa AS status,
    r.tema_zayavki AS svyazannaya_zayavka,
    t.data_sozdaniya,
    t.planovyy_srok,
    t.fakticheskiy_srok
FROM tasks t
INNER JOIN employees isp  ON t.kod_ispolnitelya = isp.kod_sotrudnika
INNER JOIN employees post ON t.kod_postanovshchika = post.kod_sotrudnika
INNER JOIN statuses s     ON t.kod_statusa = s.kod_statusa
LEFT JOIN requests r      ON t.kod_svyazannoy_zayavki = r.kod_zayavki
ORDER BY t.data_sozdaniya;

SELECT
    a.kod_soglasovaniya,
    r.tema_zayavki,
    e.fio AS soglasuyushchiy,
    d.nazvanie_podrazdeleniya AS otdel_soglasuyushchego,
    a.rezultat_soglasovaniya,
    a.kommentariy,
    a.data_resheniya
FROM approvals a
INNER JOIN requests r  ON a.kod_zayavki = r.kod_zayavki
INNER JOIN employees e ON a.kod_soglasuyushchego = e.kod_sotrudnika
INNER JOIN departments d ON e.kod_podrazdeleniya = d.kod_podrazdeleniya
ORDER BY a.data_resheniya;

SELECT
    d.kod_podrazdeleniya,
    d.nazvanie_podrazdeleniya,
    e.fio AS rukovoditel,
    e.email,
    e.telefon
FROM departments d
LEFT JOIN employees e ON d.kod_rukovoditelya = e.kod_sotrudnika
ORDER BY d.nazvanie_podrazdeleniya;

--2.3. агрегирующие операции (SUM, AVG, COUNT, MAX, MIN)
SELECT
    (SELECT COUNT(*) FROM requests)  AS vsego_zayavok,
    (SELECT COUNT(*) FROM tasks)     AS vsego_zadach,
    (SELECT COUNT(*) FROM approvals) AS vsego_soglasovaniy,
    (SELECT COUNT(*) FROM employees WHERE status_sotrudnika = 'активен') AS aktivnyh_sotrudnikov;

SELECT
    MIN(data_sozdaniya) AS samaya_rannyaya,
    MAX(data_sozdaniya) AS samaya_pozdnyaya,
    COUNT(*) AS kolichestvo
FROM requests;

SELECT
    COUNT(*) AS zakrytyh_zadach,
    ROUND(AVG(DATEDIFF(fakticheskiy_srok, data_sozdaniya)), 1) AS avg_dney,
    MIN(DATEDIFF(fakticheskiy_srok, data_sozdaniya)) AS min_dney,
    MAX(DATEDIFF(fakticheskiy_srok, data_sozdaniya)) AS max_dney
FROM tasks
WHERE fakticheskiy_srok IS NOT NULL;

--2.4. группировка GROUP BY и фильтрация HAVING
SELECT
    s.nazvanie_statusa,
    COUNT(r.kod_zayavki) AS kolichestvo
FROM statuses s
LEFT JOIN requests r ON s.kod_statusa = r.kod_statusa
GROUP BY s.kod_statusa, s.nazvanie_statusa
ORDER BY kolichestvo DESC;

SELECT
    rt.nazvanie_tipa_zayavki,
    COUNT(r.kod_zayavki) AS kolichestvo
FROM request_types rt
INNER JOIN requests r ON rt.kod_tipa_zayavki = r.kod_tipa_zayavki
GROUP BY rt.kod_tipa_zayavki, rt.nazvanie_tipa_zayavki
HAVING COUNT(r.kod_zayavki) >= 1
ORDER BY kolichestvo DESC;

SELECT
    d.nazvanie_podrazdeleniya,
    COUNT(e.kod_sotrudnika) AS kolichestvo_sotrudnikov
FROM departments d

GROUP BY d.kod_podrazdeleniya, d.nazvanie_podrazdeleniya
HAVING COUNT(e.kod_sotrudnika) >= 2
ORDER BY kolichestvo_sotrudnikov DESC;

SELECT
    e.fio AS ispolnitel,
    COUNT(t.kod_zadachi) AS kolichestvo_zadach,
    SUM(CASE WHEN t.kod_statusa = 6 THEN 1 ELSE 0 END) AS vypolneno,
    SUM(CASE WHEN t.kod_statusa <> 6 THEN 1 ELSE 0 END) AS v_rabote
FROM employees e
INNER JOIN tasks t ON e.kod_sotrudnika = t.kod_ispolnitelya
GROUP BY e.kod_sotrudnika, e.fio
HAVING COUNT(t.kod_zadachi) > 1
ORDER BY kolichestvo_zadach DESC;

--2.6. использование «функций управления потоком» (IF, CASE)
SELECT
    r.kod_zayavki,
    r.tema_zayavki,
    r.planovyy_srok,
    r.fakticheskiy_srok,
    CASE
        WHEN r.fakticheskiy_srok IS NULL AND r.planovyy_srok < CURRENT_DATE
            THEN 'ПРОСРОЧЕНА'
        WHEN r.fakticheskiy_srok IS NULL
            THEN 'В сроке (ещё открыта)'
        WHEN r.fakticheskiy_srok <= r.planovyy_srok
            THEN 'Закрыта вовремя'
        ELSE 'Закрыта с опозданием'
    END AS ocenka_sroka,
    IF(r.fakticheskiy_srok IS NULL, 'Открыта', 'Закрыта') AS sostoyanie
FROM requests r
ORDER BY r.kod_zayavki;

SELECT
    r.kod_zayavki,
    r.tema_zayavki,
    p.nazvanie_prioriteta,
    CASE p.kod_prioriteta
        WHEN 1 THEN 'Можно отложить'
        WHEN 2 THEN 'Плановая обработка'
        WHEN 3 THEN 'Срочно взять в работу'
        WHEN 4 THEN 'Немедленная реакция'
        ELSE 'Неизвестно'
    END AS rekomendaciya,
    IF(p.kod_prioriteta >= 3, 'ДА', 'НЕТ') AS trebuet_vnimaniya_rukovoditelya
FROM requests r
INNER JOIN priorities p ON r.kod_prioriteta = p.kod_prioriteta
ORDER BY p.kod_prioriteta DESC;

SELECT
    e.fio,
    e.status_sotrudnika,
    CASE e.status_sotrudnika
        WHEN 'активен'   THEN 'Полный доступ'
        WHEN 'в отпуске' THEN 'Только уведомления'
        WHEN 'уволен'    THEN 'Доступ заблокирован'
        ELSE 'Проверить вручную'
    END AS uroven_dostupa,
    IF(e.status_sotrudnika = 'активен', e.email, NULL) AS email_dlya_uvedomleniy
FROM employees e
ORDER BY e.fio;

--2.7. создание одного представления на наиболее популярный запрос (CREATE VIEW) и временное представление – оператор WITH
CREATE OR REPLACE VIEW v_zayavki_polnye AS
SELECT
    r.kod_zayavki,
    r.tema_zayavki,
    r.opisanie_zayavki,
    rt.nazvanie_tipa_zayavki AS tip,
    p.nazvanie_prioriteta    AS prioritet,
    s.nazvanie_statusa       AS status,
    e.fio                    AS initsiator,
    d.nazvanie_podrazdeleniya AS otdel_initsiatora,
    r.data_sozdaniya,
    r.planovyy_srok,
    r.fakticheskiy_srok,
    CASE
        WHEN r.fakticheskiy_srok IS NULL AND r.planovyy_srok < CURRENT_DATE THEN 'ПРОСРОЧЕНА'
        WHEN r.fakticheskiy_srok IS NULL THEN 'Открыта'
        WHEN r.fakticheskiy_srok <= r.planovyy_srok THEN 'Вовремя'
        ELSE 'С опозданием'
    END AS ocenka_sroka
FROM requests r
INNER JOIN request_types rt ON r.kod_tipa_zayavki = rt.kod_tipa_zayavki
INNER JOIN priorities p     ON r.kod_prioriteta = p.kod_prioriteta
INNER JOIN statuses s       ON r.kod_statusa = s.kod_statusa
INNER JOIN employees e      ON r.kod_initsiatora = e.kod_sotrudnika
INNER JOIN departments d    ON e.kod_podrazdeleniya = d.kod_podrazdeleniya;

SELECT * FROM v_zayavki_polnye
WHERE prioritet IN ('Высокий', 'Критический')
ORDER BY data_sozdaniya DESC;

WITH zagruzka AS (
    SELECT
        e.kod_sotrudnika,
        e.fio,
        COUNT(t.kod_zadachi) AS vsego_zadach,
        SUM(CASE WHEN t.kod_statusa = 6 THEN 1 ELSE 0 END) AS vypolneno,
        SUM(CASE WHEN t.kod_statusa <> 6 THEN 1 ELSE 0 END) AS otkryto
    FROM employees e
    INNER JOIN tasks t ON e.kod_sotrudnika = t.kod_ispolnitelya
    GROUP BY e.kod_sotrudnika, e.fio
)
SELECT
    fio,
    vsego_zadach,
    vypolneno,
    otkryto,
    CASE
        WHEN otkryto = 0 THEN 'Свободен'
        WHEN otkryto = 1 THEN 'Нормальная загрузка'
        ELSE 'Перегружен'
    END AS ocenka_zagruzki
FROM zagruzka
WHERE vsego_zadach >= 1
ORDER BY otkryto DESC, vsego_zadach DESC;

WITH bez_soglasovaniya AS (
    SELECT r.kod_zayavki, r.tema_zayavki, r.data_sozdaniya
    FROM requests r
    WHERE r.kod_zayavki NOT IN (SELECT a.kod_zayavki FROM approvals a)
)
SELECT * FROM bez_soglasovaniya
ORDER BY data_sozdaniya;
